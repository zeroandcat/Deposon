    """归一化编辑距离相似度 (v3 prereg §2.2.2 字面).

    公式: NLED-sim(a, b) = 1 - edit_distance(a, b) / max(len_a, len_b)
    范围: NLED-sim ∈ [0, 1]; 0 = 完全不对齐, 1 = 完全一致.

    m, n = len(a), len(b)
    if m == 0 and n == 0:
        return 1.0
    if m == 0 or n == 0:
        return 0.0
    d = levenshtein_distance(a, b)
    return 1.0 - d / max(m, n)
